if (/^\d+$/.test(val) && isTechMode) { isTechMode = false; document.querySelectorAll('.truck-chip').forEach(el => el.classList.remove('active')); const allBtn = document.querySelector('[data-city="all"]'); if (allBtn) allBtn.classList.add('active'); } renderCards(); autocompleteList.innerHTML = ''; if (!val) { autocompleteList.classList.add('hidden'); return; } const searchWords = val.split(/\s+/).filter(w => w.length > 0); let suggestions = []; if (isTechMode) { suggestions = (onlineTechStations.length > 0 ? onlineTechStations : techStations).filter(station => { const fullText = `${station.type} ${station.name}${station.address} ${station.desc \vert{}\vert{} ''}${station.city || ''}`.toLowerCase(); return searchWords.every(word => fullText.includes(word)); }); if (suggestions.length > 0) { autocompleteList.classList.remove('hidden'); suggestions.slice(0, 20).forEach(item => { const div = document.createElement('div'); div.className = 'px-4 py-3 hover:bg-orange-50 dark:hover:bg-slate-700 cursor-pointer border-b border-gray-50 dark:border-slate-700 last:border-none transition-colors mx-1 my-0.5 text-gray-800 dark:text-gray-200'; div.innerHTML = getDropdownItemHTML(item, true); div.addEventListener('click', () => { playTickSound(); vibrateShort(); // Вибрация searchInput.value = item.name; autocompleteList.classList.add('hidden'); searchInput.blur(); // Скрываем клавиатуру (Пункт 3) renderCards(); }); autocompleteList.appendChild(div); }); } else autocompleteList.classList.add('hidden'); } else { suggestions = stores.filter(store => { let brandName = "ЛЕНТА"; if (store.address.includes("ОКЕЙ") || store.tk === "929929") brandName = "ОКЕЙ"; else if (store.address.includes("Молл")) brandName = "РЦ МОЛЛ"; else if (store.address.includes("Монетка") || (store.tk.startsWith("92") && store.tk.length > 3)) brandName = "МОНЕТКА"; const cleanAddressTextSearch = store.address.replace(/\b\d{6}\b,?\s*/g, ''); const fullText = `тк ${store.tk} рц ${store.tk}${store.city} ${cleanAddressTextSearch}${brandName}`.toLowerCase().replace(/['"`]/g, ''); const matchesSearch = searchWords.every(word => { if (/^\d+$/.test(word)) return store.tk === word || new RegExp(`\\b${word}\\b`).test(cleanAddressTextSearch.toLowerCase()); return fullText.includes(word); }); let matchesFilter = true; if (selectedCityFilter !== 'all' && store.city !== selectedCityFilter) matchesFilter = false; return matchesSearch && matchesFilter; }); if (suggestions.length > 0) { autocompleteList.classList.remove('hidden'); suggestions.slice(0, 20).forEach(store => { const item = document.createElement('div'); item.className = 'px-4 py-3 hover:bg-blue-50 dark:hover:bg-slate-700 cursor-pointer border-b border-gray-50 dark:border-slate-700 last:border-none transition-colors mx-1 my-0.5 text-gray-800 dark:text-gray-200'; item.innerHTML = getDropdownItemHTML(store, false); item.addEventListener('click', () => { playTickSound(); vibrateShort(); // Вибрация searchInput.value = store.tk !== '-' ? store.tk : store.city; autocompleteList.classList.add('hidden'); searchInput.blur(); // Скрываем клавиатуру (Пункт 3) renderCards(); }); autocompleteList.appendChild(item); }); } else autocompleteList.classList.add('hidden'); } }); document.addEventListener('click', function(e) { if (!searchInput.contains(e.target) && !autocompleteList.contains(e.target)) { autocompleteList.classList.add('hidden'); } }); window.startVoiceSearch = function() { const micBtn = document.getElementById('voiceSearchBtn'); const searchInput = document.getElementById('searchInput'); const SpeechRecognition = window.SpeechRecognition || window.webkitSpeechRecognition; if (!SpeechRecognition) { showToast("Голосовой поиск не поддерживается 🎙"); return; } try { const recognition = new SpeechRecognition(); recognition.lang = 'ru-RU'; recognition.interimResults = false; recognition.maxAlternatives = 1; recognition.onstart = function() { micBtn.classList.add('mic-listening'); micBtn.classList.remove('text-lentaBlue', 'dark:text-blue-400'); searchInput.value = ''; searchInput.placeholder = "Слушаю вас..."; }; recognition.onresult = function(event) { let speechResult = event.results[0][0].transcript.toLowerCase(); speechResult = speechResult.replace(/\bсто\b/gi, '100').replace(/\bдвести\b/gi, '200').replace(/\bтриста\b/gi, '300').replace(/\.$/, '').trim(); }; function resetMic() { micBtn.classList.remove('mic-listening'); micBtn.classList.add('text-lentaBlue', 'dark:text-blue-400'); searchInput.placeholder = "Поиск..."; } vibrateShort(); // Вибрация при включении микрофона recognition.start(); } catch (e) { resetMic(); } };
